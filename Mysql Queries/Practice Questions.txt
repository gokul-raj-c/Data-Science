Section A — SELECT and WHERE (Q1–Q9)

1. List the name, department and salary of every employee.

select emp_name,department,salary from employees;



2. Show only the names of employees who work in the IT department.

select emp_name from employees where department='IT';



#3. Show all details of employees earning more than 60000 .

select * from employees where salary > 60000;



#4. Show the name and city of employees who are not from Mumbai .

select emp_name,city from employees where city != 'Mumbai';



#5. List employees whose salary is between 40000 and 70000 (both included).  

select emp_name from employees where salary between 40000 and 70000;



#6. List employees who work in Sales , HR or Finance . Use IN , not three OR conditions.

select emp_name from employees where department in ('HR','Sales','Finance');



#7. Find every employee whose name starts with the letter  A  then, as a second query, every employee whose name contains the letters ar anywhere. Use LIKE .

select emp_name from employees where emp_name like 'A%';
select emp_name from employees where emp_name like '%ar%';



#8. List the names of employees who have no bonus recorded, and then the names of those who do have one. Remember: = NULL does not work.

select emp_name from employees where bonus is null;
select emp_name from employees where bonus is not null;



#9. Show the name, salary and bonus of every Manager whose salary is above  60000 . Both conditions must hold

select emp_name,salary,bonus from employees where designation='Manager' and salary > 60000;









Section B — Columns, aliases, sorting, limiting (Q10–Q16)

#10. Show each employee's name and their annual salary ( salary * 12 ). Name the calculated column annual_salary .

select emp_name,salary,salary * 12 as annual_salary from employees;



#11. Show the name and the total pay of each employee, where total pay is salary plus bonus. Look at what your database returns for employees whose bonus is that happens.

select emp_name, salary+bonus as total_pay from employees;



#12. List the distinct departments present in the table. Then list the distinct cities.

select distinct department from employees;
select distinct city from employees;



#13. List all employees sorted by salary, highest first.

select emp_name,salary from employees order by salary desc;



#14. List all employees sorted by department (A→Z) and, within each department, by salary highest first.

select emp_name,department from employees order by department;



#15. Show the 3 highest-paid employees — name and salary only.

select emp_name,salary from employees order by salary desc limit 3;



#16. Show the 2 most recently hired employees.

select emp_name from employees order by hire_date desc limit 2;
