select * from employees where salary > 60000;



#4. Show the name and city of employees who are not from Mumbai .

#5. List employees whose salary is between 40000 and 70000 (both included).  

select emp_name from employees where salary between 40000 and 70000;



#6. List employees who work in Sales , HR or Finance . Use IN , not three OR conditions.

select emp_name from employees where department in ('HR','Sales','Finance');



#7. Find every employee whose name starts with the letter  A  then, as a second query, every employee whose name contains the letters ar anywhere. Use LIKE .

select emp_name from employees where emp_name like 'A%';
select emp_name from employees where emp_name like '%ar%';

#34. Show cities where the total salary paid exceeds 100000 .

select city,sum(salary) as total_salary from employees group by city having total_salary > 100000;



#35. Show designations held by exactly one employee.

select designation from employees group by designation having count(*) =1;



#36. Show departments where the average salary is above 50000 , counting only employees hired before 2022. Your query must use WHERE and HAVING together — write one line explaining which rows each of them filters.

select department, avg(salary) as average_salary from employees where year(hire_date) < 2022 group by department having average_salary > 50000;
WHERE filters individual employees hired before 2022, while HAVING filters departments whose resulting average salary is above 50,000.



#37. Using the sales table: show each product whose total quantity sold is 20 or more, listed from the largest quantity down.

40. Explain in two lines why the query below fails, then rewrite it so it works: 
SELECT department, AVG(salary) FROM employees WHERE AVG(salary) > 50000 GROUP BY department;

Why it fails (2 lines):
WHERE filters individual rows before grouping, so aggregate functions like AVG() cannot be used there.
AVG(salary) is calculated after GROUP BY, so the condition must go in HAVING.

Correct query:
SELECT department,
       AVG(salary) AS average_salary
FROM employees
GROUP BY department
HAVING AVG(salary) > 50000;
